"""Read pinned printed approximations;round independent nominal-model references.
Usage: python THIS source_revision output.json
"""
from pathlib import Path
from decimal import Decimal,Context,localcontext,ROUND_HALF_EVEN
import subprocess,sys,json,re
r=Path.cwd();pin=sys.argv[1];out=Path(sys.argv[2]);path='books/book-02-mathematical-foundations-for-finance-with-python/chapter-09/9.10-exercises-and-applied-problems.md';s=subprocess.check_output(['git','show',pin+':'+path],cwd=r,text=True)
def ex(n):return s.split('**Exercise '+str(n)+' —')[1].split('**Exercise '+str(n+1)+' —')[0]
def approx(n):return re.findall(r'\\approx(-?[0-9]+\.[0-9]+)',ex(n))
def refs(prec):
 with localcontext(Context(prec=prec,rounding=ROUND_HALF_EVEN)):
  V=Decimal(10000)*Decimal('-.2').exp();L=-5*V*Decimal('.001');correction=25*V*Decimal('.001')**2/2;Q=L+correction;E=V*(Decimal('-.005').exp()-1)-Q
  return [V,-5*V,25*V,L,correction,Q,abs(E),125*V*Decimal('.001')**3/6,Decimal('.0001')**(Decimal(1)/3)]
printed=approx(9)+approx(10)+approx(11)+[re.findall(r'0\.000[0-9]+',ex(12))[-1]]+approx(13)+approx(8);assert len(printed)==9,printed
rows=[]
for i,(text,a,b) in enumerate(zip(printed,refs(100),refs(160))):
 quantum=Decimal(1).scaleb(Decimal(text).as_tuple().exponent)
 with localcontext(Context(prec=200,rounding=ROUND_HALF_EVEN)):
  aa=a.quantize(quantum);bb=b.quantize(quantum)
 assert aa==bb
 rows.append(dict(case=['V','first derivative','second derivative','linear change','quadratic correction','quadratic change','absolute quadratic error','quadratic bound','maximum shock'][i],printed=text,expected_rounded=str(bb),reference=str(b),quantum=str(quantum),passed=Decimal(text)==bb))
d=dict(source_revision=pin,source_path=path,source_blob=subprocess.check_output(['git','rev-parse',pin+':'+path],cwd=r,text=True).strip(),predicates=rows,passed=sum(x['passed'] for x in rows),failed=sum(not x['passed'] for x in rows),reference_precisions=[100,160],rounding='nearest decimal at printed precision,ROUND_HALF_EVEN',limits='Manuscript approximation checks on exact nominal decimal model,not runtime assessment implementation or independent mathematical approval. Precision is taken from each printed answer;no tolerance widened to hide failures.')
out.write_text(json.dumps(d,indent=2)+'\n');print(json.dumps(dict(passed=d['passed'],failed=d['failed'])));raise SystemExit(d['failed']!=0)
