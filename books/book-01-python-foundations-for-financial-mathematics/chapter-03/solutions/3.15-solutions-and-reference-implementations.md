## Solutions 6–10 — Scope, Types, and Contracts

**6.** The default list is created when the definition executes and is shared by calls that omit the argument. A None default can request a fresh list inside the body. The reference append_flow below follows this contract. An explicitly supplied list is still mutated and returned; the fix does not promise purity.

**7.** Assignment makes rate local throughout that function body unless a declaration changes the binding rule. Reading it before the local assignment completes raises UnboundLocalError. Prefer adjusted_rate(rate, spread) returning their sum, with compatible rate conventions and a stated numerical policy.

**8.** Ordinary annotations do not automatically enforce runtime types, finiteness, rate domain, or compatible units. int also does not express exclusion of bool, a nonnegative horizon, or an upper limit. Decimal does not encode currency or annual/monthly meaning.

**9.** A suitable contract is: discount one finite nonnegative Decimal amount at period n to period 0 using a finite Decimal effective rate greater than -1 per matching period; require integer n from 0 through 1000 excluding bool; return a same-currency Decimal without sign reversal or explicit quantization. Validate even at zero horizon. Exclude intermediate cash flows, fees, taxes, inflation and currency/rate conversion. Apply the Section 3.14 fixed context. Raise TypeError for unsupported representations and ValueError for unsupported values; propagate trapped Decimal exceptions and reject unusable factors or non-finite results. Record finite rounding and underflow limitations.

**10.** Rebinding changes a local name, not the caller's binding. A list parameter can refer to the same mutable object held by the caller, so append changes that object. Local scope does not imply independent object ownership.

## Executable Reference Checks

Save as chapter03_solutions.py beside financial_functions.py. The list helper is an exercise-specific implementation; it does not change the project library's numerical policy or public functions.

```python
"""Selected executable solutions for Chapter 3's 20 exercises."""

from decimal import Decimal, Inexact, ROUND_DOWN, localcontext
import unittest

import financial_functions as finance


def append_flow(
    amount: Decimal,
    flows: list[Decimal] | None = None,
) -> list[Decimal]:
    """Append a finite Decimal flow to a supplied or fresh list.

    Signed flows are supported here. Omitted or None flows requests a
    fresh list. A supplied list is mutated and returned unchanged in
    identity. Existing entries are assumed validated.
    """
    if not isinstance(amount, Decimal):
        raise TypeError("amount must be Decimal")
    if not amount.is_finite():
        raise ValueError("amount must be finite")
    if flows is None:
        flows = []
    flows.append(amount)
    return flows


class TestChapter03Solutions(unittest.TestCase):
    """Verify selected implementations and applied exercise results."""

    def test_mutable_default_repair(self) -> None:
        first = append_flow(Decimal("100"))
        second = append_flow(Decimal("200"))
        self.assertEqual(first, [Decimal("100")])
        self.assertEqual(second, [Decimal("200")])
        self.assertIsNot(first, second)
        supplied: list[Decimal] = []
        self.assertIs(append_flow(Decimal("-20"), supplied), supplied)
        self.assertEqual(supplied, [Decimal("-20")])

    def test_independent_valuations(self) -> None:
        self.assertEqual(
            finance.future_value(
                Decimal("2000"), periodic_rate=Decimal("0.04"), periods=3
            ), Decimal("2249.728"),
        )
        self.assertEqual(
            finance.present_value(
                Decimal("144"), periodic_rate=Decimal("0.20"), periods=2
            ), Decimal("100"),
        )

    def test_quarterly_conversion_and_composition(self) -> None:
        rate = finance.periodic_rate_from_nominal(
            Decimal("0.08"), periods_per_year=4
        )
        self.assertEqual(rate, Decimal("0.02"))
        self.assertEqual(
            finance.effective_annual_rate(rate, periods_per_year=4),
            Decimal("0.08243216"),
        )
        wrapped = finance.present_value_from_nominal(
            Decimal("104.04"), nominal_annual_rate=Decimal("0.08"),
            periods_per_year=4, periods=2,
        )
        with localcontext() as context:
            context.prec = 40
            context.traps[Inexact] = False
            staged = Decimal("104.04") * finance.discount_factor(rate, periods=2)
            self.assertLessEqual(abs(staged - wrapped), Decimal("1e-30"))
        self.assertEqual(wrapped, Decimal("100"))

    def test_negative_rate(self) -> None:
        future = finance.future_value(
            Decimal("1000"), periodic_rate=Decimal("-0.10"), periods=1
        )
        self.assertEqual(future, Decimal("900"))
        self.assertEqual(
            finance.present_value(
                future, periodic_rate=Decimal("-0.10"), periods=1
            ), Decimal("1000"),
        )

    def test_validation_matrix(self) -> None:
        with self.assertRaises(TypeError):
            finance.present_value(
                "100", periodic_rate=Decimal("0.05"), periods=2  # type: ignore[arg-type]
            )
        with self.assertRaises(ValueError):
            finance.present_value(
                Decimal("100"), periodic_rate=Decimal("NaN"), periods=0
            )
        with self.assertRaises(TypeError):
            finance.present_value(
                Decimal("100"), periodic_rate=Decimal("0.05"), periods=True
            )
        with self.assertRaises(ValueError):
            finance.present_value(
                Decimal("100"), periodic_rate=Decimal("0.05"), periods=1001
            )

    def test_context_preservation(self) -> None:
        expected = finance.present_value(
            Decimal("1000"), periodic_rate=Decimal("0.037"), periods=17
        )
        with localcontext() as caller:
            caller.prec = 6
            caller.rounding = ROUND_DOWN
            caller.traps[Inexact] = True
            caller.flags[Inexact] = True
            flags = dict(caller.flags)
            traps = dict(caller.traps)
            settings = (
                caller.prec, caller.rounding, caller.Emin, caller.Emax,
                caller.capitals, caller.clamp,
            )
            result = finance.present_value(
                Decimal("1000"), periodic_rate=Decimal("0.037"), periods=17
            )
            self.assertEqual(result, expected)
            self.assertEqual(dict(caller.flags), flags)
            self.assertEqual(dict(caller.traps), traps)
            self.assertEqual(
                (
                    caller.prec, caller.rounding, caller.Emin, caller.Emax,
                    caller.capitals, caller.clamp,
                ), settings,
            )

    def test_final_integration(self) -> None:
        monthly = finance.periodic_rate_from_nominal(
            Decimal("0.12"), periods_per_year=12
        )
        annual = finance.effective_annual_rate(monthly, periods_per_year=12)
        future = finance.future_value(
            Decimal("1000"), periodic_rate=monthly, periods=12
        )
        recovered = finance.present_value_from_nominal(
            future, nominal_annual_rate=Decimal("0.12"),
            periods_per_year=12, periods=12,
        )
        self.assertEqual(monthly, Decimal("0.01"))
        self.assertEqual(annual, Decimal("0.126825030131969720661201"))
        self.assertEqual(future, Decimal("1126.825030131969720661201"))
        with localcontext() as context:
            context.prec = 40
            context.traps[Inexact] = False
            self.assertLessEqual(abs(recovered - Decimal("1000")), Decimal("1e-30"))
        self.assertEqual(
            finance.present_value_from_nominal(
                Decimal("1000"), nominal_annual_rate=Decimal("0.12"),
                periods_per_year=12, periods=0,
            ), Decimal("1000"),
        )
        with self.assertRaises(ValueError):
            finance.present_value_from_nominal(
                Decimal("1000"), nominal_annual_rate=Decimal("0.12"),
                periods_per_year=0, periods=0,
            )


if __name__ == "__main__":
    unittest.main()
```
